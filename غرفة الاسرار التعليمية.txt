from js import document

class MobilePythonGame:
    def __init__(self):
        self.puzzles = [
            {
                "question": "ألغاز الرياضيات: ما هو حاصل ضرب 7 × 8؟",
                "options": ["54", "56", "64"],
                "correct": 1
            },
            {
                "question": "ألغاز المنطق: التتابع 2, 4, 8, 16.. ما الرقم التالي؟",
                "options": ["24", "30", "32"],
                "correct": 2
            },
            {
                "question": "ألغاز العلوم: ما هو الغاز الأساسي لتنفس الكائنات الحية؟",
                "options": ["الأكسجين", "النيتروجين", "ثاني أكسيد الكربون"],
                "correct": 0
            }
        ]
        self.current_step = 0

    def render(self):
        if self.current_step < len(self.puzzles):
            p = self.puzzles[self.current_step]
            document.getElementById("question").innerText = f"اللغز ({self.current_step + 1}/3):\n{p['question']}"
            
            opts_html = ""
            for idx, opt in enumerate(p["options"]):
                opts_html += f'<button py-click="game.check_answer({idx})">{opt}</button>'
            
            document.getElementById("options").innerHTML = opts_html
        else:
            document.getElementById("question").innerText = "🎉 مبروك! لقد قمت بفك جميع الألغاز بنجاح والهروب من الغرفة!"
            document.getElementById("options").innerHTML = '<button py-click="game.reset()">إعادة اللعب</button>'
            document.getElementById("output").innerText = ""

    def check_answer(self, choice):
        p = self.puzzles[self.current_step]
        if choice == p["correct"]:
            document.getElementById("output").style.color = "#4ade80"
            document.getElementById("output").innerText = "✅ إجابة صحيحة! أحسنت."
            self.current_step += 1
        else:
            document.getElementById("output").style.color = "#f87171"
            document.getElementById("output").innerText = "❌ إجابة خاطئة! حاول مرة أخرى بتركيز."
        
        self.render()

    def reset(self):
        self.current_step = 0
        document.getElementById("output").innerText = ""
        self.render()

# تشغيل اللعبة
game = MobilePythonGame()
game.render()
